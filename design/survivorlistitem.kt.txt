/**
 * Tactical triage item representing a survivor discovered via gossip packet propagation.
 * Displays urgency badge (with icon + text), sector tag, hop count, encrypted medical status indicator,
 * and time since last broadcast.
 * Has an accessible minimum height and touch target >= 56dp.
 */
@Composable
fun SurvivorListItem(
    survivor: SurvivorPacket,
    onClick: () -> Unit,
    modifier: Modifier = Modifier
) {
    val minutesAgo = ((System.currentTimeMillis() - survivor.timestamp) / 60000L).coerceAtLeast(1)

    val urgencyBorderColor = when (survivor.statusTag) {
        UrgencyStatus.TRAPPED -> EmergencyRed
        UrgencyStatus.INJURED -> EmergencyOrange
        UrgencyStatus.NEEDS_INSULIN -> EmergencyYellow
        UrgencyStatus.NEED_WATER -> Color(0xFF3898FF)
        UrgencyStatus.SAFE -> EmergencyGreen
    }

    Card(
        colors = CardDefaults.cardColors(containerColor = ColorSurfaceDark),
        shape = RoundedCornerShape(10.dp),
        border = androidx.compose.foundation.BorderStroke(1.dp, ColorSurfaceBorder),
        modifier = modifier
            .fillMaxWidth()
            .clickable(role = Role.Button) { onClick() }
            .semantics {
                contentDescription = "Survivor ${survivor.survivorName}, status ${survivor.statusTag.displayName}, sector ${survivor.sectorCode}, ${survivor.hopCount} hops away, reported $minutesAgo minutes ago"
            }
    ) {
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(14.dp),
            verticalAlignment = Alignment.CenterVertically
        ) {
            // Status bar indicator line on left
            Box(
                modifier = Modifier
                    .width(4.dp)
                    .height(64.dp)
                    .clip(RoundedCornerShape(2.dp))
                    .background(urgencyBorderColor)
            )

            Spacer(modifier = Modifier.width(12.dp))

            Column(modifier = Modifier.weight(1f)) {
                // Header with Name and Time
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Text(
                        text = survivor.survivorName,
                        style = MaterialTheme.typography.titleSmall,
                        fontWeight = FontWeight.Bold,
                        color = TextPrimary
                    )

                    Text(
                        text = stringResource(R.string.survivor_time_format, minutesAgo),
                        style = MaterialTheme.typography.labelSmall,
                        color = TextMuted
                    )
                }

                Spacer(modifier = Modifier.height(6.dp))

                // Status chip & sector tag
                Row(
                    verticalAlignment = Alignment.CenterVertically,
                    horizontalArrangement = Arrangement.spacedBy(8.dp)
                ) {
                    StatusChip(
                        status = survivor.statusTag,
                        isSelected = true
                    )

                    Surface(
                        shape = RoundedCornerShape(4.dp),
                        color = ColorSurfaceElevatedDark
                    ) {
                        Text(
                            text = survivor.sectorCode,
                            style = MaterialTheme.typography.labelSmall,
                            fontWeight = FontWeight.Bold,
                            color = MeshCyan,
                            modifier = Modifier.padding(horizontal = 6.dp, vertical = 4.dp)
                        )
                    }
                }

                Spacer(modifier = Modifier.height(8.dp))

                // Hop count & Encrypted Medical Status
                Row(
                    verticalAlignment = Alignment.CenterVertically,
                    horizontalArrangement = Arrangement.spacedBy(12.dp)
                ) {
                    Row(verticalAlignment = Alignment.CenterVertically) {
                        Icon(
                            imageVector = Icons.Default.NearMe,
                            contentDescription = null,
                            tint = TextMuted,
                            modifier = Modifier.size(14.dp)
                        )
                        Spacer(modifier = Modifier.width(4.dp))
                        Text(
                            text = if (survivor.hopCount <= 0) {
                                stringResource(R.string.survivor_direct_hop)
                            } else {
                                stringResource(R.string.survivor_hops_format, survivor.hopCount)
                            },
                            style = MaterialTheme.typography.bodySmall,
                            color = TextSecondary
                        )
                    }

                    Row(verticalAlignment = Alignment.CenterVertically) {
                        Icon(
                            imageVector = if (survivor.medicalData.isEncrypted) Icons.Default.Lock else Icons.Default.LockOpen,
                            contentDescription = null,
                            tint = if (survivor.medicalData.isEncrypted) MeshCyan else EmergencyGreen,
                            modifier = Modifier.size(14.dp)
                        )
                        Spacer(modifier = Modifier.width(4.dp))
                        Text(
                            text = if (survivor.medicalData.isEncrypted) {
                                stringResource(R.string.survivor_encrypted_med)
                            } else {
                                stringResource(R.string.survivor_unlocked_med)
                            },
                            style = MaterialTheme.typography.bodySmall,
                            color = if (survivor.medicalData.isEncrypted) TextSecondary else EmergencyGreen
                        )
                    }
                }
            }

            Spacer(modifier = Modifier.width(8.dp))

            // Navigation forward chevron
            Icon(
                imageVector = Icons.AutoMirrored.Filled.ArrowForward,
                contentDescription = null,
                tint = TextMuted,
                modifier = Modifier.size(20.dp)
            )
        }
    }
}
