package com.meshguard.app.ui.screens.survivor

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.meshguard.app.data.model.HistoryNode
import com.meshguard.app.data.model.MeshPeer
import com.meshguard.app.data.repository.FakeMeshRepository
import com.meshguard.app.data.repository.MeshRepository
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.combine
import kotlinx.coroutines.flow.stateIn
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

/**
 * Log entry detailing gossip packet relay events and BLE GATT sync handshakes.
 */
data class MeshSyncLogEntry(
    val id: String,
    val timestampFormatted: String,
    val peerAlias: String,
    val eventType: String,
    val packetCount: Int,
    val rssi: Int,
    val isOutbound: Boolean
)

data class MeshNetworkUiState(
    val peers: List<MeshPeer> = emptyList(),
    val packetsCarriedCount: Int = 28,
    val isBroadcasting: Boolean = true,
    val isBatterySaverActive: Boolean = true,
    val breadcrumbs: List<HistoryNode> = emptyList(),
    val syncLogs: List<MeshSyncLogEntry> = emptyList(),
    val isScanning: Boolean = false
)

class MeshNetworkViewModel(
    private val meshRepository: MeshRepository = FakeMeshRepository()
) : ViewModel() {

    private val timeFormatter = SimpleDateFormat("HH:mm:ss", Locale.getDefault())

    private val _syncLogs = MutableStateFlow(
        listOf(
            MeshSyncLogEntry(
                id = "log-1",
                timestampFormatted = timeFormatter.format(Date(System.currentTimeMillis() - 4000L)),
                peerAlias = "Mule Relay #04",
                eventType = "BLE GATT SYNC COMPLETE",
                packetCount = 6,
                rssi = -64,
                isOutbound = false
            ),
            MeshSyncLogEntry(
                id = "log-2",
                timestampFormatted = timeFormatter.format(Date(System.currentTimeMillis() - 11000L)),
                peerAlias = "Search Team Echo",
                eventType = "DISTRESS BEACON FORWARDED",
                packetCount = 1,
                rssi = -72,
                isOutbound = true
            ),
            MeshSyncLogEntry(
                id = "log-3",
                timestampFormatted = timeFormatter.format(Date(System.currentTimeMillis() - 35000L)),
                peerAlias = "Citizen Node #C9",
                eventType = "DATA MULE REPLICATION",
                packetCount = 4,
                rssi = -86,
                isOutbound = false
            ),
            MeshSyncLogEntry(
                id = "log-4",
                timestampFormatted = timeFormatter.format(Date(System.currentTimeMillis() - 62000L)),
                peerAlias = "Medic Field Station",
                eventType = "TRIAGE QUEUE ACKNOWLEDGED",
                packetCount = 12,
                rssi = -59,
                isOutbound = true
            ),
            MeshSyncLogEntry(
                id = "log-5",
                timestampFormatted = timeFormatter.format(Date(System.currentTimeMillis() - 120000L)),
                peerAlias = "Shelter Gate 2",
                eventType = "PEER CONTACT ESTABLISHED",
                packetCount = 5,
                rssi = -91,
                isOutbound = false
            )
        )
    )

    private val _isScanning = MutableStateFlow(false)

    val uiState: StateFlow<MeshNetworkUiState> = combine(
        meshRepository.nearbyPeers,
        meshRepository.packetsCarriedCount,
        meshRepository.isBroadcastingBeacon,
        meshRepository.isBatterySaverActive,
        meshRepository.breadcrumbChain,
        _syncLogs,
        _isScanning
    ) { peers, packets, broadcasting, eco, breadcrumbs, logs, scanning ->
        MeshNetworkUiState(
            peers = peers,
            packetsCarriedCount = packets,
            isBroadcasting = broadcasting,
            isBatterySaverActive = eco,
            breadcrumbs = breadcrumbs,
            syncLogs = logs,
            isScanning = scanning
        )
    }.stateIn(
        scope = viewModelScope,
        started = SharingStarted.WhileSubscribed(5000),
        initialValue = MeshNetworkUiState()
    )

    fun onToggleBroadcast() {
        meshRepository.toggleBroadcast()
    }

    fun onForceRescan() {
        viewModelScope.launch {
            _isScanning.value = true
            delay(1500) // Simulate radio BLE discovery window
            _isScanning.value = false

            val newLog = MeshSyncLogEntry(
                id = "log-${System.currentTimeMillis()}",
                timestampFormatted = timeFormatter.format(Date()),
                peerAlias = "Radio Discovery Engine",
                eventType = "SCAN CYCLE COMPLETED (${uiState.value.peers.size} PEERS FOUND)",
                packetCount = 0,
                rssi = -60,
                isOutbound = false
            )
            _syncLogs.update { listOf(newLog) + it.take(19) }
        }
    }

    fun onPeerClicked(peer: MeshPeer) {
        // Record inspection or trigger manual direct sync handshake
        val log = MeshSyncLogEntry(
            id = "log-${System.currentTimeMillis()}",
            timestampFormatted = timeFormatter.format(Date()),
            peerAlias = peer.alias,
            eventType = "MANUAL P2P PING ACKNOWLEDGED",
            packetCount = 1,
            rssi = peer.rssi,
            isOutbound = true
        )
        _syncLogs.update { listOf(log) + it.take(19) }
    }
}
