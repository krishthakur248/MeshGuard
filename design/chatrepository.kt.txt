package com.meshguard.app.data.repository

import com.meshguard.app.data.model.ChatMessage
import com.meshguard.app.data.model.MessageType
import kotlinx.coroutines.flow.StateFlow

/**
 * Interface managing offline hop-by-hop mesh messaging and 15s voice-drop relay notes.
 */
interface ChatRepository {
    val messages: StateFlow<List<ChatMessage>>

    fun sendTextMessage(content: String)
    fun sendVoiceDropMessage(audioDurationSec: Int, voiceBase64Payload: String)
    fun markDeliveredToRescuer(messageId: String)
}
