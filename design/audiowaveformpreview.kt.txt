/**
 * 15-second emergency voice memo waveform preview and player control.
 * Has minimum 56dp accessible tap targets, dual visual cues (play icon + duration label),
 * and audio wave amplitude bars.
 */
@Composable
fun AudioWaveformPreview(
    durationSec: Int,
    isPlaying: Boolean,
    onTogglePlay: () -> Unit,
    modifier: Modifier = Modifier,
    playedProgressFraction: Float = 0.35f,
    activeColor: Color = MeshCyan
) {
    // 20 normalized amplitude bar heights representing a 15-second voice snippet
    val waveAmplitudes = listOf(
        0.30f, 0.45f, 0.70f, 0.90f, 0.55f,
        0.40f, 0.85f, 1.00f, 0.65f, 0.35f,
        0.50f, 0.75f, 0.95f, 0.60f, 0.40f,
        0.80f, 0.50f, 0.35f, 0.60f, 0.25f
    )

    val playActionDesc = if (isPlaying) {
        stringResource(R.string.audio_pause_desc)
    } else {
        stringResource(R.string.audio_play_desc)
    }

    Row(
        modifier = modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(8.dp))
            .background(ColorSurfaceDark)
            .border(1.dp, ColorSurfaceBorder, RoundedCornerShape(8.dp))
            .padding(8.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        // Minimum 56dp play/pause target
        IconButton(
            onClick = onTogglePlay,
            modifier = Modifier
                .size(56.dp)
                .clip(CircleShape)
                .background(ColorSurfaceElevatedDark)
                .semantics {
                    contentDescription = "$playActionDesc ($durationSec seconds)"
                    role = Role.Button
                }
        ) {
            Icon(
                imageVector = if (isPlaying) Icons.Default.Pause else Icons.Default.PlayArrow,
                contentDescription = null,
                tint = activeColor,
                modifier = Modifier.size(28.dp)
            )
        }

        Spacer(modifier = Modifier.width(12.dp))

        // Waveform bars with progress coloration
        Column(
            modifier = Modifier.weight(1f)
        ) {
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .height(28.dp),
                verticalAlignment = Alignment.CenterVertically,
                horizontalArrangement = Arrangement.spacedBy(3.dp)
            ) {
                waveAmplitudes.forEachIndexed { index, amplitude ->
                    val barFraction = index.toFloat() / waveAmplitudes.size
                    val isBarPlayed = barFraction <= playedProgressFraction

                    Box(
                        modifier = Modifier
                            .weight(1f)
                            .height((28.dp * amplitude).coerceAtLeast(4.dp))
                            .clip(RoundedCornerShape(2.dp))
                            .background(if (isBarPlayed) activeColor else ColorSurfaceBorder)
                    )
                }
            }

            Spacer(modifier = Modifier.height(4.dp))

            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Text(
                    text = stringResource(R.string.audio_memo_label),
                    style = MaterialTheme.typography.labelSmall,
                    color = TextSecondary
                )
                Text(
                    text = stringResource(R.string.audio_duration_format, durationSec),
                    style = MaterialTheme.typography.labelSmall,
                    fontFamily = FontFamily.Monospace,
                    fontWeight = FontWeight.Bold,
                    color = activeColor
                )
            }
        }
    }
}

@Preview(showBackground = true)
@Composable
fun AudioWaveformPreviewPreview() {
    Surface(color = ColorBackgroundDark) {
        Box(modifier = Modifier.padding(16.dp)) {
            AudioWaveformPreview(
                durationSec = 14,
                isPlaying = false,
                onTogglePlay = {},
                playedProgressFraction = 0.40f
            )
        }
    }
}
