/**
 * Route Composable connecting the HomeViewModel and navigation actions.
 */
@Composable
fun HomeRoute(
    onNavigateToStatusPicker: () -> Unit,
    onNavigateToChat: () -> Unit,
    onNavigateToMedicalId: () -> Unit,
    onNavigateToMeshNetwork: () -> Unit,
    viewModel: HomeViewModel = viewModel()
) {
    val uiState by viewModel.uiState.collectAsState()

    HomeScreen(
        uiState = uiState,
        onToggleBeacon = viewModel::onToggleBeacon,
        onToggleBatterySaver = viewModel::onToggleBatterySaver,
        onStatusClicked = onNavigateToStatusPicker,
        onChatClicked = onNavigateToChat,
        onMedicalIdClicked = onNavigateToMedicalId,
        onMeshNetworkClicked = onNavigateToMeshNetwork
    )
}

/**
 * Stateless HomeScreen:
 * 1. Persistent MeshTopBar (peer count, packets carried, eco indicator)
 * 2. Status Selector Tile with high-contrast dual-cue chip & quick edit action
 * 3. Centered Pulsing BeaconButton (accessible tap target >= 190dp)
 * 4. Live Mesh Stats Row (Active peers / Packets carried / Last sync)
 * 5. Battery Saver ECO Mode Toggle Card (>= 56dp)
 * 6. Tactical Quick Access Grid: Chat Messages & Encrypted Medical ID
 */
@Composable
fun HomeScreen(
    uiState: HomeUiState,
    onToggleBeacon: () -> Unit,
    onToggleBatterySaver: (Boolean) -> Unit,
    onStatusClicked: () -> Unit,
    onChatClicked: () -> Unit,
    onMedicalIdClicked: () -> Unit,
    onMeshNetworkClicked: () -> Unit,
    modifier: Modifier = Modifier
) {
    val scrollState = rememberScrollState()

    Surface(
        modifier = modifier.fillMaxSize(),
        color = ColorBackgroundDark
    ) {
        Column(modifier = Modifier.fillMaxSize()) {
            // Persistent Mesh Guard Top App Bar
            MeshTopBar(
                title = stringResource(R.string.app_name).uppercase(),
                peerCount = uiState.activePeerCount,
                packetsCarried = uiState.packetsCarriedCount,
                isBatterySaverActive = uiState.isBatterySaverActive
            )

            Column(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxWidth()
                    .verticalScroll(scrollState)
                    .padding(horizontal = 20.dp, vertical = 12.dp),
                horizontalAlignment = Alignment.CenterHorizontally
            ) {
                // Sector & Current Urgency Status Bar
                CurrentStatusBar(
                    status = uiState.myStatus,
                    sectorCode = uiState.sectorCode,
                    onClick = onStatusClicked
                )

                Spacer(modifier = Modifier.height(20.dp))

                // Centered Beacon Button with accessible pulsing ripple ring
                BeaconButton(
                    status = uiState.myStatus,
                    isBroadcasting = uiState.isBroadcastingBeacon,
                    onClick = onToggleBeacon,
                    modifier = Modifier.padding(vertical = 4.dp)
                )

                Spacer(modifier = Modifier.height(8.dp))

                // Beacon status subtitle with dual cues
                Text(
                    text = if (uiState.isBroadcastingBeacon) {
                        "OFFLINE CHIRP BROADCAST ACTIVE"
                    } else {
                        "BEACON PAUSED (SILENT RELAY ONLY)"
                    },
                    style = MaterialTheme.typography.labelSmall,
                    fontFamily = FontFamily.Monospace,
                    fontWeight = FontWeight.Bold,
                    color = if (uiState.isBroadcastingBeacon) MeshCyan else TextMuted,
                    letterSpacing = 1.sp
                )

                Spacer(modifier = Modifier.height(20.dp))

                // Tactical Mesh Stats Row (Peers, Mule Packets, Sync Latency)
                MeshStatsRow(
                    peerCount = uiState.activePeerCount,
                    packetsCarried = uiState.packetsCarriedCount,
                    lastSyncSec = uiState.lastSyncAgoSec,
                    onClick = onMeshNetworkClicked
                )

                Spacer(modifier = Modifier.height(16.dp))

                // Battery Saver Toggle Card (>= 56dp touch target)
                BatterySaverCard(
                    isEcoModeActive = uiState.isBatterySaverActive,
                    onToggle = onToggleBatterySaver
                )

                Spacer(modifier = Modifier.height(16.dp))

                // Quick Navigation Grid (Chat Thread & Encrypted Medical ID)
                QuickAccessSection(
                    unreadMessages = uiState.unreadMessageCount,
                    onChatClicked = onChatClicked,
                    onMedicalIdClicked = onMedicalIdClicked
                )

                Spacer(modifier = Modifier.height(16.dp))
            }
        }
    }
}

/**
 * Interactive current status display card opening the StatusPicker screen.
 */
@Composable
private fun CurrentStatusBar(
    status: UrgencyStatus,
    sectorCode: String,
    onClick: () -> Unit,
    modifier: Modifier = Modifier
) {
    Card(
        colors = CardDefaults.cardColors(containerColor = ColorSurfaceDark),
        shape = RoundedCornerShape(12.dp),
        border = androidx.compose.foundation.BorderStroke(1.dp, ColorSurfaceBorder),
        modifier = modifier
            .fillMaxWidth()
            .clickable(role = Role.Button) { onClick() }
            .semantics {
                contentDescription = "Current emergency status is ${status.displayName}, located in sector $sectorCode. Tap to change status."
            }
    ) {
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 14.dp, vertical = 12.dp),
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.SpaceBetween
        ) {
            Row(
                verticalAlignment = Alignment.CenterVertically,
                modifier = Modifier.weight(1f)
            ) {
                StatusChip(
                    status = status,
                    isSelected = true
                )

                Spacer(modifier = Modifier.width(10.dp))

                Column {
                    Text(
                        text = "LOCATION SECTOR",
                        style = MaterialTheme.typography.labelSmall,
                        color = TextMuted,
                        letterSpacing = 1.sp
                    )
                    Text(
                        text = sectorCode,
                        style = MaterialTheme.typography.titleSmall,
                        fontFamily = FontFamily.Monospace,
                        fontWeight = FontWeight.Bold,
                        color = MeshCyan
                    )
                }
            }

            Row(verticalAlignment = Alignment.CenterVertically) {
                Text(
                    text = "CHANGE",
                    style = MaterialTheme.typography.labelMedium,
                    fontWeight = FontWeight.Bold,
                    color = MeshCyan
                )
                Spacer(modifier = Modifier.width(4.dp))
                Icon(
                    imageVector = Icons.Default.ChevronRight,
                    contentDescription = null,
                    tint = MeshCyan,
                    modifier = Modifier.size(18.dp)
                )
            }
        }
    }
}

/**

private fun MeshStatsRow(
    peerCount: Int,
    packetsCarried: Int,
    lastSyncSec: Int,
    onClick: () -> Unit,
    modifier: Modifier = Modifier
) {
    Card(
        colors = CardDefaults.cardColors(containerColor = ColorSurfaceDark),
        shape = RoundedCornerShape(12.dp),
        border = androidx.compose.foundation.BorderStroke(1.dp, ColorSurfaceBorder),
        modifier = modifier
            .fillMaxWidth()
            .clickable(role = Role.Button) { onClick() }
            .semantics {
                contentDescription = "Mesh telemetry: $peerCount peers in range, $packetsCarried packets carried, last synced $lastSyncSec seconds ago. Tap for network topology."
            }
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(14.dp)
        ) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Row(verticalAlignment = Alignment.CenterVertically) {
                    Icon(
                        imageVector = Icons.Default.Hub,
                        contentDescription = null,
                        tint = MeshCyan,
                        modifier = Modifier.size(16.dp)
                    )
                    Spacer(modifier = Modifier.width(6.dp))
                    Text(
                        text = "LOCAL MESH TELEMETRY",
                        style = MaterialTheme.typography.labelSmall,
                        fontWeight = FontWeight.Bold,
                        letterSpacing = 1.sp,
                        color = TextSecondary
                    )
                }

                Row(verticalAlignment = Alignment.CenterVertically) {
                    Text(
                        text = "TOPOLOGY",
                        style = MaterialTheme.typography.labelSmall,
                        color = MeshCyan,
                        fontWeight = FontWeight.Bold
                    )
                    Icon(
                        imageVector = Icons.Default.ChevronRight,
                        contentDescription = null,
                        tint = MeshCyan,
                        modifier = Modifier.size(14.dp)
                    )
                }
            }

            Spacer(modifier = Modifier.height(12.dp))

            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceAround,
                verticalAlignment = Alignment.CenterVertically
            ) {
                // Metric 1: Peers
                StatMetricColumn(
                    icon = Icons.Default.Radio,
                    iconTint = if (peerCount > 0) EmergencyGreen else EmergencyOrange,
                    value = peerCount.toString(),
                    label = "Active Peers"
                )

                // Divider
                Box(
                    modifier = Modifier
                        .height(36.dp)
                        .width(1.dp)
                        .background(ColorSurfaceBorder)
                )

                // Metric 2: Data Mule Packets
                StatMetricColumn(
                    icon = Icons.Default.SwapHoriz,
                    iconTint = MeshCyan,
                    value = packetsCarried.toString(),
                    label = "Pkts Carried"
                )

                // Divider
                Box(
                    modifier = Modifier
                        .height(36.dp)
                        .width(1.dp)
                        .background(ColorSurfaceBorder)
                )

                // Metric 3: Last Sync
                StatMetricColumn(
                    icon = Icons.Default.Sync,
                    iconTint = TextSecondary,
                    value = "${lastSyncSec}s",
                    label = "Last Contact"
                )
            }
        }
    }
}

@Composable
private fun StatMetricColumn(
    icon: ImageVector,
    iconTint: Color,
    value: String,
    label: String
) {
    Column(horizontalAlignment = Alignment.CenterHorizontally) {
        Row(
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.spacedBy(4.dp)
        ) {
            Icon(
                imageVector = icon,
                contentDescription = null,
                tint = iconTint,
                modifier = Modifier.size(14.dp)
            )
            Text(
                text = value,
                style = MaterialTheme.typography.titleMedium,
                fontFamily = FontFamily.Monospace,
                fontWeight = FontWeight.Black,
                color = TextPrimary
            )
        }
        Spacer(modifier = Modifier.height(2.dp))
        Text(
            text = label,
            style = MaterialTheme.typography.labelSmall,
            color = TextMuted
        )
    }
}

/**
 * Battery Saver card with high-contrast tactical styling and accessible touch target >= 56dp.
 */
@Composable
private fun BatterySaverCard(
    isEcoModeActive: Boolean,
    onToggle: (Boolean) -> Unit,
    modifier: Modifier = Modifier
) {
    Card(
        colors = CardDefaults.cardColors(containerColor = ColorSurfaceElevatedDark),
        shape = RoundedCornerShape(12.dp),
        border = androidx.compose.foundation.BorderStroke(
            1.dp,
            if (isEcoModeActive) EmergencyYellow.copy(alpha = 0.5f) else ColorSurfaceBorder
        ),
        modifier = modifier
            .fillMaxWidth()
            .semantics {
                contentDescription = "Battery-Saver ECO mode is ${if (isEcoModeActive) "active" else "disabled"}. Duty cycles BLE radio intervals to extend phone battery up to 72 hours."
            }
    ) {
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .height(64.dp) // Accessible height >= 56dp
                .padding(horizontal = 16.dp),
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.SpaceBetween
        ) {
            Row(
                verticalAlignment = Alignment.CenterVertically,
                modifier = Modifier.weight(1f)
            ) {
                Box(
                    modifier = Modifier
                        .size(36.dp)
                        .clip(CircleShape)
                        .background(if (isEcoModeActive) EmergencyYellow.copy(alpha = 0.15f) else ColorSurfaceDark),
                    contentAlignment = Alignment.Center
                ) {
                    Icon(
                        imageVector = Icons.Default.BatterySaver,
                        contentDescription = null,
                        tint = if (isEcoModeActive) EmergencyYellow else TextMuted,
                        modifier = Modifier.size(20.dp)
                    )
                }

                Spacer(modifier = Modifier.width(12.dp))

                Column {
                    Row(verticalAlignment = Alignment.CenterVertically) {
                        Text(
                            text = "ECO BATTERY CYCLE",
                            style = MaterialTheme.typography.labelMedium,
                            fontWeight = FontWeight.Bold,
                            color = TextPrimary
                        )
                        if (isEcoModeActive) {
                            Spacer(modifier = Modifier.width(6.dp))
                            Surface(
                                shape = RoundedCornerShape(4.dp),
                                color = EmergencyYellow.copy(alpha = 0.2f)
                            ) {
                                Text(
                                    text = "72h TARGET",
                                    style = MaterialTheme.typography.labelSmall,
                                    fontFamily = FontFamily.Monospace,
                                    fontWeight = FontWeight.Bold,
                                    color = EmergencyYellow,
                                    modifier = Modifier.padding(horizontal = 6.dp, vertical = 2.dp)
                                )
                            }
                        }
                    }
                    Text(
                        text = "5s broadcast / 25s sleep interval",
                        style = MaterialTheme.typography.bodySmall,
                        color = TextSecondary
                    )
                }
            }

            Switch(
                checked = isEcoModeActive,
                onCheckedChange = onToggle,
                colors = SwitchDefaults.colors(
                    checkedThumbColor = Color.Black,
                    checkedTrackColor = EmergencyYellow,
                    uncheckedThumbColor = TextMuted,
                    uncheckedTrackColor = ColorSurfaceDark
                )
            )
        }
    }
}

/**
 * 2-Card quick navigation section for Emergency Chat thread and Encrypted Medical ID.
 */
@Composable
private fun QuickAccessSection(
    unreadMessages: Int,
    onChatClicked: () -> Unit,
    onMedicalIdClicked: () -> Unit,
    modifier: Modifier = Modifier
) {
    Row(
        modifier = modifier.fillMaxWidth(),
        horizontalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        // Quick Action 1: Mesh Messages
        Card(
            colors = CardDefaults.cardColors(containerColor = ColorSurfaceDark),
            shape = RoundedCornerShape(12.dp),
            border = androidx.compose.foundation.BorderStroke(1.dp, ColorSurfaceBorder),
            modifier = Modifier
                .weight(1f)
                .height(84.dp) // Accessible >= 56dp
                .clickable(role = Role.Button) { onChatClicked() }
                .semantics {
                    contentDescription = "Open Mesh Messages. $unreadMessages crisis broadcasts available."
                }
        ) {
            Column(
                modifier = Modifier
                    .fillMaxSize()
                    .padding(12.dp),
                verticalArrangement = Arrangement.SpaceBetween
            ) {
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Box(
                        modifier = Modifier
                            .size(32.dp)
                            .clip(CircleShape)
                            .background(ColorSurfaceElevatedDark),
                        contentAlignment = Alignment.Center
                    ) {
                        Icon(
                            imageVector = Icons.Default.ChatBubble,
                            contentDescription = null,
                            tint = MeshCyan,
                            modifier = Modifier.size(16.dp)
                        )
                    }

                    if (unreadMessages > 0) {
                        Surface(
                            shape = CircleShape,
                            color = EmergencyRed
                        ) {
                            Text(
                                text = unreadMessages.toString(),
                                style = MaterialTheme.typography.labelSmall,
                                fontWeight = FontWeight.Black,
                                color = Color.White,
                                modifier = Modifier.padding(horizontal = 6.dp, vertical = 2.dp)
                            )
                        }
                    }
                }

                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.Bottom
                ) {
                    Text(
                        text = "Mesh Chat",
                        style = MaterialTheme.typography.titleSmall,
                        fontWeight = FontWeight.Bold,
                        color = TextPrimary
                    )
                    Icon(
                        imageVector = Icons.AutoMirrored.Filled.ArrowForward,
                        contentDescription = null,
                        tint = TextMuted,
                        modifier = Modifier.size(16.dp)
                    )
                }
            }
        }

        // Quick Action 2: Encrypted Medical ID
        Card(
            colors = CardDefaults.cardColors(containerColor = ColorSurfaceDark),
            shape = RoundedCornerShape(12.dp),
            border = androidx.compose.foundation.BorderStroke(1.dp, ColorSurfaceBorder),
            modifier = Modifier
                .weight(1f)
                .height(84.dp) // Accessible >= 56dp
                .clickable(role = Role.Button) { onMedicalIdClicked() }
                .semantics {
                    contentDescription = "Open Encrypted Medical ID. Contains allergies, blood type, and conditions."
                }
        ) {
            Column(
                modifier = Modifier
                    .fillMaxSize()
                    .padding(12.dp),
                verticalArrangement = Arrangement.SpaceBetween
            ) {
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Box(
                        modifier = Modifier
                            .size(32.dp)
                            .clip(CircleShape)
                            .background(ColorSurfaceElevatedDark),
                        contentAlignment = Alignment.Center
                    ) {
                        Icon(
                            imageVector = Icons.Default.MedicalServices,
                            contentDescription = null,
                            tint = EmergencyYellow,
                            modifier = Modifier.size(16.dp)
                        )
                    }

                    Surface(
                        shape = RoundedCornerShape(4.dp),
                        color = EmergencyGreen.copy(alpha = 0.2f)
                    ) {
                        Text(
                            text = "AES-256",
                            style = MaterialTheme.typography.labelSmall,
                            fontFamily = FontFamily.Monospace,
                            fontWeight = FontWeight.Bold,
                            color = EmergencyGreen,
                            modifier = Modifier.padding(horizontal = 4.dp, vertical = 2.dp)
                        )
                    }
                }

                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.Bottom
                ) {
                    Text(
                        text = "Medical ID",
                        style = MaterialTheme.typography.titleSmall,
                        fontWeight = FontWeight.Bold,
                        color = TextPrimary
                    )
                    Icon(
                        imageVector = Icons.AutoMirrored.Filled.ArrowForward,
                        contentDescription = null,
                        tint = TextMuted,
                        modifier = Modifier.size(16.dp)
                    )
                }
            }
        }
    }
}
